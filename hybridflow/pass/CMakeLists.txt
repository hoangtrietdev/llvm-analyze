cmake_minimum_required(VERSION 3.16)
project(HFLoopAnalysis C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(LLVM REQUIRED CONFIG)
message(STATUS "Found LLVM ${LLVM_PACKAGE_VERSION} in ${LLVM_DIR}")
include_directories(SYSTEM ${LLVM_INCLUDE_DIRS})
separate_arguments(LLVM_DEFS NATIVE_COMMAND "${LLVM_DEFINITIONS}")
add_definitions(${LLVM_DEFS})
# DependenceInfo::depends() lost its PossiblyLoopIndependent flag in newer LLVM.
foreach(dir ${LLVM_INCLUDE_DIRS})
  if(EXISTS "${dir}/llvm/Analysis/DependenceAnalysis.h")
    file(READ "${dir}/llvm/Analysis/DependenceAnalysis.h" DAH)
    if(DAH MATCHES "bool PossiblyLoopIndependent")
      add_definitions(-DHF_DA_POSSIBLY_LOOP_INDEPENDENT=1)
      message(STATUS "DA API: depends(Src, Dst, PossiblyLoopIndependent)")
    endif()
  endif()
endforeach()
if(NOT LLVM_ENABLE_RTTI)
  set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -fno-rtti")
endif()
add_library(HFLoopAnalysis MODULE HFLoopAnalysis.cpp)
if(APPLE)
  # Symbols are resolved against the host `opt` binary at load time.
  target_link_options(HFLoopAnalysis PRIVATE -undefined dynamic_lookup)
endif()
